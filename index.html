const ROOT_FOLDER_INPUT = '請在此貼上你的Google_Drive總資料夾網址或ID';

// 保留預設 myFunction 入口，直接按頂部「▶ 執行」即可運作，不會再報錯！
function myFunction() {
  scanScatteredDriveFiles();
}

function scanScatteredDriveFiles() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ss.toast('正在掃描資料夾內的零散 Word / Excel 檔案，請稍候...', '🤖 傲翔自動抓取器', 30);

  const rootFolder = resolveRootFolder(ss);
  const mainSheet = ss.getSheets()[0];
  mainSheet.setName('幼童統整總表');
  let logSheet = ss.getSheetByName('流程表紀錄') || ss.insertSheet('流程表紀錄');

  const studentMap = {}; // 以「幼童姓名」作為唯一鎖匙拼合所有零散檔案
  const flowchartRows = [['幼童姓名', '日期', '訓練模式', '流程/目標內容', '進度代號', '來源檔案']];

  const stopWords = ['姓名','幼童姓名','學生姓名','性別','男','女','班別','負責老師','班主任','備註','特殊需要','日期','時間','目標','語言','認知','自理','小肌肉','大肌肉','社交'];
  const ensureStudent = (name) => {
    const clean = String(name || '').replace(/\s+/g, '').trim();
    if (!clean || clean.length < 2 || clean.length > 4 || stopWords.includes(clean)) return null;
    if (!studentMap[clean]) {
      studentMap[clean] = {
        name: clean, gender: '', group: '', className: '', teacher: '',
        dob: '', tier: '', mode: '', statusTag: '', fileNo: '', remarks: '', specialNeeds: '',
        goals: { 語言: [], 社交情緒: [], 認知: [], 小肌肉: [], 自理: [], 大肌肉: [], 綜合目標: [] }
      };
    }
    return studentMap[clean];
  };

  // 遞迴掃描總資料夾及所有子資料夾
  function walkFolder(folder) {
    const files = folder.getFiles();
    while (files.hasNext()) {
      const file = files.next();
      const fname = file.getName();
      if (file.getId() === ss.getId() || fname.startsWith('~$') || fname.startsWith('_temp_')) continue;

      try {
        // A. 抓取《班別名單.docx》
        if (fname.includes('名單')) {
          const text = readDocText(file);
          let currentGroup = '方換 — 下年名單';
          text.split(/\r?\n/).map(l => l.trim()).filter(Boolean).forEach(line => {
            if (line.includes('T1')) currentGroup = '方換 — T1 名單';
            else if (line.includes('慧妍')) currentGroup = '慧妍';
            else if (line.includes('蔡繼有')) currentGroup = '蔡繼有';
            else if (line.includes('下年名單')) currentGroup = '方換 — 下年名單';

            const parts = line.split(/\t|\s{2,}|\|/).map(s => s.trim()).filter(Boolean);
            const nameHit = parts.find(p => /^[\u4e00-\u9fa5]{2,4}$/.test(p) && !stopWords.includes(p));
            if (nameHit && parts.length >= 2) {
              const st = ensureStudent(nameHit);
              if (st) {
                if (!st.group) st.group = currentGroup;
                const cls = parts.find(p => /(上|下)?(高|低|幼)[A-D]?/.test(p));
                if (cls && !st.className) st.className = cls;
                const tchr = parts.find(p => p.includes('老師') || p.includes('主任'));
                if (tchr && !st.teacher) st.teacher = tchr;
                const tierHit = parts.find(p => /^(A|B|T1)$/i.test(p));
                if (tierHit && !st.tier) st.tier = tierHit.toUpperCase();
                const modeHit = parts.find(p => /(入班|個別|小組)/.test(p));
                if (modeHit && !st.mode) st.mode = modeHit;
              }
            }
          });
        }
        // B. 抓取《學童資料.xlsx》或試算表（出生日期、特殊需要、檔案編號）
        else if (fname.includes('學童資料')) {
          const rows = readSpreadsheetRows(file);
          rows.forEach(r => {
            const rowStr = r.join(' | ');
            const nameCandidate = r.find(c => /^[\u4e00-\u9fa5]{2,4}$/.test(String(c).trim()) && !stopWords.includes(String(c).trim()));
            if (!nameCandidate) return;
            const st = ensureStudent(nameCandidate);
            if (!st) return;
            if (r.some(c => String(c).trim() === '男')) st.gender = '男';
            else if (r.some(c => String(c).trim() === '女')) st.gender = '女';
            const dobMatch = rowStr.match(/(20\d{2}[-/.]\d{1,2}[-/.]\d{1,2})/);
            if (dobMatch) st.dob = dobMatch[1].replace(/[/.]/g, '-');
            const icfMatch = rowStr.match(/(ICF\/[A-Za-z0-9&\/_-]+)/);
            if (icfMatch) st.fileNo = icfMatch[1];
            const longDiag = r.filter(c => String(c).length > 8 && !String(c).includes('ICF/')).join('；');
            if (longDiag) st.specialNeeds = longDiag;
          });
        }
        // C. 抓取各份《ITP_姓名.docx》個別學習計劃（六大範疇目標）
        else if (fname.includes('ITP') || fname.includes('個別訓練計劃') || fname.includes('個別學習計劃')) {
          const text = readDocText(file);
          const nameFromTitle = fname.match(/(?:ITP|計劃)[_＿\s]*(?:parent[_＿\s]*)?([\u4e00-\u9fa5]{2,4})/i);
          const nameFromBody = text.match(/(?:姓名|學生姓名|幼童姓名)[：:\s]*([\u4e00-\u9fa5]{2,4})/);
          const childName = (nameFromTitle && nameFromTitle[1]) || (nameFromBody && nameFromBody[1]);
          const st = ensureStudent(childName);
          if (st) {
            let currentDomain = '綜合目標';
            text.split(/\r?\n/).map(l => l.trim()).filter(Boolean).forEach(line => {
              if (/語言/.test(line) && line.length < 20) currentDomain = '語言';
              else if (/社交|情緒/.test(line) && line.length < 20) currentDomain = '社交情緒';
              else if (/認知/.test(line) && line.length < 20) currentDomain = '認知';
              else if (/小肌/.test(line) && line.length < 20) currentDomain = '小肌肉';
              else if (/自理/.test(line) && line.length < 20) currentDomain = '自理';
              else if (/大肌/.test(line) && line.length < 20) currentDomain = '大肌肉';
              if (line.includes('能') && line.length >= 6) {
                if (!st.goals[currentDomain].includes(line)) st.goals[currentDomain].push(line);
              }
            });
          }
        }
        // D. 抓取各份《流程記錄表.docx》（統計目標執行次數）
        else if (fname.includes('流程')) {
          const text = readDocText(file);
          const modeType = fname.includes('小組') ? '小組' : (fname.includes('個別') ? '個別' : '入班');
          const lines = text.split(/\r?\n/).map(l => l.trim()).filter(Boolean);
          let activeChild = '';
          let activeDate = '';
          lines.forEach(line => {
            Object.keys(studentMap).forEach(n => { if (line.includes(n)) activeChild = n; });
            const dMatch = line.match(/(\d{1,2}\/\d{1,2})/);
            if (dMatch) activeDate = dMatch[1];
            if (activeChild && line.includes('能') && line.length >= 6) {
              flowchartRows.push([activeChild, activeDate || '本學期', modeType, line, '', fname]);
            }
          });
        }
      } catch (err) {
        Logger.log('略過檔案 ' + fname + ': ' + err.message);
      }
    }
    const subFolders = folder.getFolders();
    while (subFolders.hasNext()) walkFolder(subFolders.next());
  }

  walkFolder(rootFolder);

  // 寫入《幼童統整總表》分頁
  const headers = ['編號','幼童姓名','性別','所屬組別','班別','負責老師','出生日期','實齡','支援層級','訓練模式','狀態標籤','中心檔案編號','排堂及特別備註','特殊需要與能力概況','語言目標','社交/情緒目標','認知目標','小肌肉目標','自理目標','大肌肉目標','綜合目標'];
  const outRows = [headers];
  Object.values(studentMap).forEach((s, i) => {
    outRows.push([
      i + 1, s.name, s.gender, s.group || '方換 — 下年名單', s.className, s.teacher, s.dob, '',
      s.tier, s.mode, s.statusTag, s.fileNo, s.remarks, s.specialNeeds,
      s.goals['語言'].join('；'), s.goals['社交情緒'].join('；'), s.goals['認知'].join('；'),
      s.goals['小肌肉'].join('；'), s.goals['自理'].join('；'), s.goals['大肌肉'].join('；'), s.goals['綜合目標'].join('；')
    ]);
  });
  mainSheet.clearContents();
  mainSheet.getRange(1, 1, outRows.length, headers.length).setValues(outRows);

  // 寫入《流程表紀錄》分頁
  logSheet.clearContents();
  logSheet.getRange(1, 1, flowchartRows.length, 6).setValues(flowchartRows);

  const summaryMsg = '抓取完成！共統整 ' + (outRows.length - 1) + ' 位幼童及 ' + (flowchartRows.length - 1) + ' 節流程紀錄。';
  Logger.log(summaryMsg);
  ss.toast(summaryMsg, '✅ 完成', 15);
}

// 自動解析資料夾網址或 ID（若留空則自動讀取試算表所在資料夾）
function resolveRootFolder(ss) {
  const raw = String(ROOT_FOLDER_INPUT || '').trim();
  if (raw && !raw.includes('請在此貼上')) {
    const m = raw.match(/\/folders\/([a-zA-Z0-9_-]+)/);
    const folderId = m ? m[1] : raw.split('?')[0].trim();
    return DriveApp.getFolderById(folderId);
  }
  const parents = DriveApp.getFileById(ss.getId()).getParents();
  if (parents.hasNext()) return parents.next();
  throw new Error('請在第 2 行 ROOT_FOLDER_INPUT 貼上你的 Google Drive 總資料夾網址或 ID！');
}

// 同時兼容 Drive API v3 (預設) 與 v2 的轉檔輔助函式
function copyOfficeToTempGoogleFile(fileId, targetMime) {
  if (typeof Drive === 'undefined') {
    throw new Error('請先在左側「服務 (Services) ＋」揀選「Drive API」並按新增，再按執行！');
  }
  try {
    return Drive.Files.copy({ name: '_temp_plk_convert', mimeType: targetMime }, fileId);
  } catch (e) {
    return Drive.Files.copy({ title: '_temp_plk_convert', mimeType: targetMime }, fileId);
  }
}

function readSpreadsheetRows(file) {
  if (file.getMimeType() === 'application/vnd.google-apps.spreadsheet') {
    return SpreadsheetApp.openById(file.getId()).getSheets()[0].getDataRange().getDisplayValues();
  }
  const temp = copyOfficeToTempGoogleFile(file.getId(), 'application/vnd.google-apps.spreadsheet');
  try {
    return SpreadsheetApp.openById(temp.id).getSheets()[0].getDataRange().getDisplayValues();
  } finally {
    DriveApp.getFileById(temp.id).setTrashed(true);
  }
}

function readDocText(file) {
  if (file.getMimeType() === 'application/vnd.google-apps.document') {
    return DocumentApp.openById(file.getId()).getBody().getText();
  }
  const temp = copyOfficeToTempGoogleFile(file.getId(), 'application/vnd.google-apps.document');
  try {
    const body = DocumentApp.openById(temp.id).getBody();
    const lines = [body.getText()];
    body.getTables().forEach(t => {
      for (let r = 0; r < t.getNumRows(); r++) {
        const rowCells = [];
        for (let c = 0; c < t.getRow(r).getNumCells(); c++) {
          rowCells.push(t.getRow(r).getCell(c).getText().replace(/\s+/g, ' ').trim());
        }
        lines.push(rowCells.join(' | '));
      }
    });
    return lines.join('\n');
  } finally {
    DriveApp.getFileById(temp.id).setTrashed(true);
  }
}
